// ==========================================
// CLASS #1 - PERSON
// ==========================================

class Person {

    // ENCAPSULATION #1
    #name;

    constructor(name) {
        this.#name = name;
    }

    // METHOD #1
    getName() {
        return this.#name;
    }

    // METHOD #2
    introduce() {
        return "My name is " + this.#name;
    }
}


// ==========================================
// CLASS #2 - CUSTOMER
// INHERITANCE #1
// ==========================================

class Customer extends Person {

    constructor(name, equipment) {
        super(name);
        this.equipment = equipment;
    }

    // METHOD #3
    rentEquipment() {

        // CONDITIONAL #1
        if (this.equipment !== "") {

            return this.getName() +
                " rented the " +
                this.equipment;

        } else {

            return "No equipment selected";
        }
    }
}


// ==========================================
// CLASS #3 - STAFF
// INHERITANCE #2
// ==========================================

class Staff extends Person {

    constructor(name, position) {
        super(name);
        this.position = position;
    }

    // METHOD #4
    assistCustomer() {

        return this.getName() +
            " is assisting the customer.";
    }

    // POLYMORPHISM
    introduce() {

        return this.getName() +
            " works at Hope's Photography Rentals.";
    }
}


// ==========================================
// CLASS #4 - RENTAL SHOP
// ==========================================

class RentalShop {

    // ENCAPSULATION #2
    #equipment;

    constructor(equipment) {
        this.#equipment = equipment;
    }

    // METHOD #5
    showEquipment() {

        console.log("\nPHOTOGRAPHY EQUIPMENT");

        // LOOP #1
        for (let item of this.#equipment) {

            console.log(
                "- " + item
            );
        }
    }

    // METHOD #6
    checkEquipment(itemName) {

        // LOOP #2
        for (let item of this.#equipment) {

            // CONDITIONAL #2
            if (
                item.toLowerCase() ===
                itemName.toLowerCase()
            ) {

                return item +
                    " is available for rent.";
            }
        }

        return itemName +
            " is not available.";
    }

    // METHOD #7
    addEquipment(itemName) {

        // CONDITIONAL #3
        if (itemName !== "") {

            this.#equipment.push(itemName);

            return itemName +
                " was added to the equipment list.";

        } else {

            return "Equipment name cannot be empty.";
        }
    }
}


// ==========================================
// ABSTRACTION
// ==========================================

// The availability-checking process is hidden
// inside this function.

function rentItem(customer, item, shop) {

    let result = shop.checkEquipment(item);

    if (result.includes("available")) {

        console.log(
            customer.rentEquipment()
        );

    } else {

        console.log(
            "Rental failed: " + result
        );
    }
}
